% --- VERSNUMMER ---
% Numren finns ALLTID i textfilerna. Om de visas styrs av installningar.tex.
% Versnumret har NOLLBREDD och kostar därför inga sidor. Det skjuts åt
% vänster in i ordmellanrummet före versen och lyfts upp, så att det inte
% lägger sig ovanpå ordet efter. Justeras med SRBversnummerVanster/Lyft.
% V26: källtexten skriver alltid "\versenumber{N}~Ord" - dvs ett hårt
% mellanslag (~) kommer OMEDELBART efter. Lägg därför INGEN egen \hspace
% här, annars blir det dubbla mellanrum. Numret är grått, som tilläggsorden.
% V27: INGEN fast bredd på numret. En fast högerställd ruta gjorde att
% breda tal (särskilt tvåsiffriga) svämmade över åt VÄNSTER och åt upp det
% mellanslag som redan finns i källtexten före numret - då såg det ut som
% om numret satt klistrat vid föregående ord/skiljetecken. Nu får numret
% sin naturliga bredd, med en liten egen luft på var sida (kan vara 0).
% V29: mellanrummet EFTER numret styrs helt av OSS (\srbvsp), inte av
% källtextens "~" (hårt mellanslag). Polyglossias svenska språkstöd ger "~"
% ett mellanrum som är ungefär DUBBELT så brett som ett vanligt ordmellanrum
% - synligt som glapp runt varenda versnummer. \srbvsp har en känd,
% kontrollerad bredd oavsett språkinställningar.
% V29c: numret + luften på båda sidor byggs som EN stel \hbox. Fristående
% \hspace/\kern mellan ord kan annars fångas upp av radens marginaljustering
% och sträckas olika mycket beroende på hur mycket den enskilda raden behöver
% fyllas ut - det gav oförutsägbart varierande glapp som inte gick att
% justera bort genom att bara ändra längden. En hbox är stel: dess innehåll
% påverkas aldrig av omgivande radjustering.
% \SRBgrad är TEXT ("8.3pt"), inte ett dimensionsregister - därför kan man
% inte skriva 0.6\SRBgrad. Vi lägger värdet i ett riktigt register först.
\newdimen\srbGradDim
\AtBeginDocument{\srbGradDim=\SRBgrad}
\newcommand{\srbVerseShow}[1]{%
  % V37: storlek och lyft är PROPORTIONELLA mot brödtexten (\SRBgrad), inte
  % absoluta. Tidigare var lyftet angivet i "ex", vilket mäts i den OMGIVANDE
  % fonten - vid liten brödtext blev lyftet därför så stort att siffran
  % hamnade uppe vid raden OVANFÖR och såg ut att höra dit.
  %   storlek = SRBversnummerAndel  × brödtextens grad  (typiskt 0.60)
  %   lyft    = SRBversnummerLyftAndel × brödtextens grad (typiskt 0.34)
  % \smash nollar boxens höjd/djup så att den upphöjda siffran inte ökar
  % radavståndet - annars kostar versnumren ~160 sidor i hela bibeln.
  \leavevmode\hbox{\kern\SRBversnummerVanster
    \smash{\raisebox{\SRBversnummerLyftAndel\srbGradDim}{%
      \textcolor{versnummerfarg}{%
        \fontsize{\SRBversnummerAndel\srbGradDim}{\SRBversnummerAndel\srbGradDim}%
        \addfontfeatures{Numbers=Lining}\selectfont #1}}}%
    \kern\SRBversnummerHoger}%
}

% V29b: RIGID mellanrum (utan plus/minus). Elastiskt \hspace deltar annars i
% radens marginaljustering och kan svälja en oproportionerligt stor del av
% radens sträckning - det var den verkliga orsaken till de ojämna glappen,
% inte längden vi angav.
% V38: \ignorespaces äter upp källtextens mellanslag efter numret. Det
% mellanslaget är ELASTISKT och sträcks av marginaljusteringen, vilket gjorde
% att avståndet mellan siffra och ord varierade kraftigt från rad till rad.
% Nu styr enbart vår egen fasta \kern (SRBversnummerHoger) - lika på alla rader.
\newcommand{\srbvsp}{\ignorespaces}

\newcommand{\ProcessAndRenderFile}[3][]{%
  % 1) Read the entire file into \temptext
  \CatchFileDef{\temptext}{#2}{}%
  % 2) Extract the very first letter
  \StrLeft{\temptext}{1}[\firstletter]%
  % 3) Remove that first letter from the text
  \StrGobbleLeft{\temptext}{1}[\resttemp]%
  % 4) Split \resttemp into first word + remainder
  \StrBefore{\resttemp}{ }[\firstword]%
  \StrBehind{\resttemp}{ }[\remainingtext]%
  % -- Add entry to the TOC using the optional argument if provided;
  %    otherwise, fallback to the heading text (#3):
  \phantomsection
  \addcontentsline{toc}{subsection}{\ifx&#1&\relax\protect\mbox{#3}\else\protect\mbox{#1}\fi}%
  % -- Print the actual heading in the document as HUGE with line breaks, etc.:
  % V14: ett kapitel får inte börja så nära sidfoten att anfangen hänger ned
  % i notzonen. Kräver plats för rubrik + anfangens höjd innan brytning.
  \Needspace*{\SRBkapitelminplats\baselineskip}%
  \InsertMark{srbch}{#3}%
  \InsertMark{srbbok}{\srbCurrentBook}% V17: boknamnet, för sidfoten
  \InsertMark{srbnr}{\srbCurrentNr}%   V17: kapitelnumret, för sidfoten
  % V31: rubrikstil styrs av SRBrubrikstil (installningar.tex). Sex alternativ:
  %   kursiv   = standard, som tidigare
  %   sparrad  = upprätt fet, breddad bokstavsspärrning
  %   linjer   = upprätt fet, tunna linjer på var sida
  %   ram      = upprätt fet, inramad
  %   initial  = stor accentfärgad förstabokstav, resten upprätt kapitäler
  %   tunn     = upprätt, VANLIG vikt (ej fet), mycket breddad spärrning - luftig
  \def\srbRubTxt{\addfontfeatures{Numbers=Lining}\MakeUppercase{#3}}
  \ifx\SRBrubrikstil\srbRubrikSparrad
    \subsection*{\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\bfseries\addfontfeatures{LetterSpace=-2.2}\srbRubTxt}%
  \else\ifx\SRBrubrikstil\srbRubrikLinjer
    \subsection*{%
      \noindent\hbox to 0.14\linewidth{\leaders\hrule height 0.5pt\hfil}\hspace{0.8em}%
      {\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\bfseries\srbRubTxt}\hspace{0.8em}%
      \hbox to 0.14\linewidth{\leaders\hrule height 0.5pt\hfil}}%
  \else\ifx\SRBrubrikstil\srbRubrikRam
    \subsection*{\centering
      \fboxsep6pt\fbox{\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\bfseries\srbRubTxt}}%
  \else\ifx\SRBrubrikstil\srbRubrikInitial
    \subsection*{%
      {\fontsize{28pt}{28pt}\selectfont\bfseries\color{titlecolor}\MakeUppercase{\srbHeadFirstLetter{#3}}}%
      {\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\bfseries\addfontfeatures{Numbers=Lining,LetterSpace=-2.2}\MakeUppercase{\srbHeadRestLetters{#3}}}}%
  \else\ifx\SRBrubrikstil\srbRubrikTunn
    \subsection*{\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\addfontfeatures{LetterSpace=-2.2}\srbRubTxt}%
  \else
    \subsection*{\fontsize{\SRBrubrikAndel\srbGradDim}{\SRBrubrikAndel\srbGradDim}\selectfont\addfontfeatures{FakeBold=\SRBrubrikfetning}\itshape\srbRubTxt}%
  \fi\fi\fi\fi\fi
  
  % -- If first letter is "J", use custom lettrine settings; otherwise use default:
  % Original settings: 0.21 on J and 0.3 on the rest
  % V12: anfangens inställningar väljs FÖRST; texten placeras utanför \IfStrEq
  % (xstring-argument är korta och tål inte \par från styckebrytningen)
  \IfStrEq{\firstletter}{J}%
    {\def\srbLettOpt{loversize=\SRBanfangJ, lines=\SRBanfangrader, lraise=\SRBanfangJlyft}}%
    {\def\srbLettOpt{loversize=\SRBanfangOvriga, lines=\SRBanfangrader, lraise=\SRBanfangOvrigalyft}}%
  \expandafter\lettrine\expandafter[\srbLettOpt]{\firstletter}{\srbFirstWord{\firstword}} \remainingtext
  \par
  \vspace{\SRBkapitelluft}
  % \backtotoc % TILLBAKA-KNAPPEN FINNS HÄR
}
